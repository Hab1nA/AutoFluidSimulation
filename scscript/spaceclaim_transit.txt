def Main():

    # /ScriptArgs 传入格式: "构型名,STEP文件路径,SCDOC输出目录"
    try:
        file_index = int(args[0])
        if file_index <= 0:
            raise ValueError("参数必须是大于0的整数")
    except Exception as e:
        print(f"参数错误: {e}")
        return

    print(f"正在处理: {step_path}")

    if not os.path.exists(step_path):
        print(f"错误: 文件 {step_path} 不存在")
        return

    # 3. 打开文档
    doc = Document.Open(step_path, None)
    part = doc.MainPart

    # 4. 几何处理逻辑 - 创建命名选择集
    primarySelection = PowerSelection.Faces.ByAreaRange(MM2(8.55), MM2(8.56), PowerSelectOptions(False, Body1))

    primarySelection = PowerSelection.Faces.ByAreaRange(MM2(9.54), MM2(9.55), PowerSelectOptions(False, Body1))

    primarySelection = PowerSelection.Faces.ByAreaRange(MM2(17222), MM2(17223), PowerSelectOptions(False, Body1))
    secondarySelection = Selection.Empty()

    primarySelection = PowerSelection.Faces.ByAreaRange(MM2(28520), MM2(28521), PowerSelectOptions(False, Body1))
    secondarySelection = Selection.Empty()

    window = Window.ActiveWindow
    if window:
